cmake_minimum_required(VERSION 3.25)
project(Screen VERSION 1.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /WX /permissive- /utf-8 /guard:cf)
  add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT)
endif()

add_library(screen_core INTERFACE)
target_include_directories(screen_core INTERFACE src)

include(CTest)
if(BUILD_TESTING)
  add_executable(screen_tests tests/core_tests.cpp)
  target_link_libraries(screen_tests PRIVATE screen_core)
  add_test(NAME screen_core COMMAND screen_tests)
endif()

if(WIN32)
  enable_language(RC)
  add_executable(Screen WIN32 src/main.cpp src/overlay.cpp src/desktop.cpp src/virtual_desktops.cpp src/settings.cpp src/ui.cpp src/preferences.cpp src/startup.cpp resources/Screen.rc)
  target_link_libraries(Screen PRIVATE screen_core comctl32 dwmapi shell32 wtsapi32 ole32 uuid advapi32 Magnification windowsapp)
  target_compile_definitions(Screen PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_options(Screen PRIVATE /MANIFEST:NO)
  if(BUILD_TESTING)
    add_executable(screen_windows_tests tests/windows_tests.cpp src/settings.cpp src/virtual_desktops.cpp)
    target_link_libraries(screen_windows_tests PRIVATE screen_core ole32 shell32 advapi32 uuid dwmapi)
    target_compile_definitions(screen_windows_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    add_test(NAME screen_windows COMMAND screen_windows_tests)
  endif()
  add_executable(ScreenStartup WIN32 src/startup_entry.cpp)
  target_compile_definitions(ScreenStartup PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  install(TARGETS Screen RUNTIME DESTINATION .)
endif()
